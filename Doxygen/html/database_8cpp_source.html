297 QString column;
298 switch (task_number) {
299 case 1: column = "task1_stat"; break;
300 case 2: column = "task2_stat"; break;

313 query.prepare(sql_request);
314 query.bindValue(":points", points);
315 query.bindValue(":socket_id", socket_descriptor);
