78QString get_task(QStringList args, int socket_descriptor) {
79 Database* db = Database::getInstance();
80 if(args.size() < 2) {
81 return "get_task_error\r\n";
82 }
83 if (!db->is_connected(socket_descriptor)) {
84 return "not_authenticated\r\n";
85 }
86 int task_type = args[1].toInt();
87 if (task_type < 1 || task_type > 5) {
88 return "task_not_found\r\n";
89 }
90 CurrentTask current = db->get_current_task(socket_descriptor);
91 if (current.is_valid() && current.type == task_type) {
92 return QString("TASK %1 %2\r\n").arg(current.type).arg(current.payload);
93 }
94 CurrentTask new_task = generate_task(task_type);
95 if (!new_task.is_valid()) {
96 return "task_generation_error\r\n";
97 }
98 if (!db->save_current_task(socket_descriptor, new_task)) {
99 return "task_save_error\r\n";
100 }
101 return QString("TASK %1 %2\r\n").arg(new_task.type).arg(new_task.payload);
102}

109QString solve_task(QStringList args, int socket_descriptor) {
110 Database* db = Database::getInstance();
111 CurrentTask task = db->get_current_task(socket_descriptor);
112
113 if (args.size() >= 3 && db->is_connected(socket_descriptor) && task.is_valid()) {
114 int task_number = args[1].toInt();
115 QString user_answer = args[2];
116
117 if (task_number < 1 || task_number > 5 || task_number != task.type) {
118 return "task_not_found\r\n";
119 }
120
121 bool is_correct = false;
122
123 if (task_number == 4) {
124 is_correct = (user_answer == task.answer);
125 } else {
126 bool ok1 = false, ok2 = false;
127 double a = user_answer.toDouble(&ok1);
128 double b = task.answer.toDouble(&ok2);
129 if (ok1 && ok2) {
130 is_correct = std::abs(a - b) < 1e-3;
131 }
132 }
133
134 db->update_stat(task_number, socket_descriptor, is_correct);
135 db->clear_current_task(socket_descriptor);
136 return is_correct ? "answer_correct\r\n" : "answer_incorrect\r\n";
137 }
138 return "solve_task_error\r\n";
139}
